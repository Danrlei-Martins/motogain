<script>
/* =============================================================
   MOTOGAIN — Sistema de Gestão para Motoristas de Aplicativo
   Arquitetura: OOP (Classes + Services) + LocalStorage
   Estrutura pronta para integração com API REST Java/Spring Boot

const Utils = {
  brl(v) {
    return new Intl.NumberFormat('pt-BR', { style: 'currency', currency: 'BRL' }).format(v || 0);
  },
  dec(v) { return (v || 0).toFixed(2).replace('.', ','); },
  mesAtual() { return new Date().toISOString().slice(0, 7); },
  hoje() { return new Date().toISOString().slice(0, 10); },
  uid() { return Date.now().toString(36) + Math.random().toString(36).slice(2); },
  dataFormatada(str) {
    if (!str) return '—';
    const [y, m, d] = str.split('-');
    return `${d}/${m}/${y}`;
  },
  mesAnoFormatado(str) {
    if (!str) return '—';
    const [y, m] = str.split('-');
    const meses = ['Jan','Fev','Mar','Abr','Mai','Jun','Jul','Ago','Set','Out','Nov','Dez'];
    return `${meses[parseInt(m)-1]} / ${y}`;
  },
  clampPct(v) { return Math.min(100, Math.max(0, v)); },
  /** Retorna todos os dias do mês no formato YYYY-MM-DD até o dia atual (ou fim do mês se mês passado) */
  diasDoMes(mes) {
    const hoje = Utils.hoje();
    const [y, m] = mes.split('-').map(Number);
    const diasNoMes = new Date(y, m, 0).getDate();
    const days = [];
    for (let d = 1; d <= diasNoMes; d++) {
      const ds = `${y}-${String(m).padStart(2,'0')}-${String(d).padStart(2,'0')}`;
      if (ds <= hoje) days.push(ds);
      else break;
    }
    return days;
  },
  /** Número de dias úteis (trabalhados) que já passaram, dado o total de dias planejados no mês */
  diasDecorridos(mes) {
    const hoje = Utils.hoje();
    const [y, m] = mes.split('-').map(Number);
    const diasNoMes = new Date(y, m, 0).getDate();
    const diaAtual = parseInt(hoje.slice(8, 10));
    const mesAtual = hoje.slice(0, 7);
    if (mes > mesAtual) return 0;
    if (mes < mesAtual) return diasNoMes;
    return diaAtual;
  }
};

class StorageService {
  constructor(prefix = 'motogain_') { this.prefix = prefix; }
  getList(key) {
    try { return JSON.parse(localStorage.getItem(this.prefix + key) || '[]'); } catch { return []; }
  }
  setList(key, data) { localStorage.setItem(this.prefix + key, JSON.stringify(data)); }
  addItem(key, item) {
    const list = this.getList(key);
    list.unshift(item);
    this.setList(key, list);
    return list;
  }
  removeItem(key, id) {
    const list = this.getList(key).filter(i => i.id !== id);
    this.setList(key, list);
    return list;
  }
  getObj(key) {
    try { return JSON.parse(localStorage.getItem(this.prefix + key) || '{}'); } catch { return {}; }
  }
  setObj(key, data) { localStorage.setItem(this.prefix + key, JSON.stringify(data)); }
}

class GanhosService {
  constructor(storage) { this.storage = storage; this.KEY = 'ganhos'; }
  salvar(ganho) { return this.storage.addItem(this.KEY, { id: Utils.uid(), ...ganho }); }
  remover(id) { return this.storage.removeItem(this.KEY, id); }
  listar() { return this.storage.getList(this.KEY); }
  porMes(mes) { return this.listar().filter(g => g.data && g.data.startsWith(mes)); }
  porDia(data) { return this.listar().filter(g => g.data === data); }
  faturamentoPorDia(data) {
    return this.porDia(data).reduce((s, g) =>
      s + (parseFloat(g.ganhoUber)||0) + (parseFloat(g.ganho99)||0) + (parseFloat(g.ganhoOutros)||0), 0);
  }
  totaisMes(mes) {
    const ganhos = this.porMes(mes);
    return ganhos.reduce((acc, g) => ({
      faturamentoBruto: acc.faturamentoBruto + (parseFloat(g.ganhoUber)||0) + (parseFloat(g.ganho99)||0) + (parseFloat(g.ganhoOutros)||0),
      totalCorridas: acc.totalCorridas + (parseInt(g.corridasUber)||0) + (parseInt(g.corridas99)||0) + (parseInt(g.corridasOutros)||0),
      totalHoras: acc.totalHoras + (parseFloat(g.horas)||0),
      totalKmTrab: acc.totalKmTrab + (parseFloat(g.kmTrab)||0),
    }), { faturamentoBruto: 0, totalCorridas: 0, totalHoras: 0, totalKmTrab: 0 });
  }
}

class DespesasService {
  constructor(storage) {
    this.storage = storage;
    this.KEY_AB = 'abastecimentos';
    this.KEY_D  = 'despesas';
  }
  salvarAbastecimento(ab) { return this.storage.addItem(this.KEY_AB, { id: Utils.uid(), ...ab }); }
  removerAbastecimento(id) { return this.storage.removeItem(this.KEY_AB, id); }
  listarAbastecimentos() { return this.storage.getList(this.KEY_AB); }
  salvarDespesa(d) { return this.storage.addItem(this.KEY_D, { id: Utils.uid(), ...d }); }
  removerDespesa(id) { return this.storage.removeItem(this.KEY_D, id); }
  listarDespesas() { return this.storage.getList(this.KEY_D); }
  totalDespesasMes(mes) {
    const abs = this.listarAbastecimentos().filter(a => a.data && a.data.startsWith(mes))
      .reduce((s, a) => s + (parseFloat(a.valor)||0), 0);
    const desps = this.listarDespesas().filter(d => d.data && d.data.startsWith(mes))
      .reduce((s, d) => s + (parseFloat(d.valor)||0), 0);
    return { totalAbastecimentos: abs, totalOutrasDespesas: desps, total: abs + desps };
  }
  calcularMediaKmL(veiculo, kmAtual) {
    const hist = this.listarAbastecimentos()
      .filter(a => a.veiculo === veiculo)
      .sort((a, b) => a.km - b.km);
    if (!hist.length) return null;
    const anterior = hist[hist.length - 1];
    const kmRodado  = kmAtual - parseFloat(anterior.km);
    const litrosRef = parseFloat(anterior.litros);
    if (kmRodado <= 0 || litrosRef <= 0) return null;
    return { media: kmRodado / litrosRef, kmRodado, litrosRef, anterior };
  }
}

class MetasService {
  constructor(storage) { this.storage = storage; this.KEY = 'metas'; }
  salvar(mes, faturamento, lucro, dias) {
    const metas = this.storage.getObj(this.KEY);
    metas[mes] = {
      faturamento: parseFloat(faturamento) || 0,
      lucro: parseFloat(lucro) || 0,
      dias: parseInt(dias) || 0
    };
    this.storage.setObj(this.KEY, metas);
  }
  porMes(mes) {
    const metas = this.storage.getObj(this.KEY);
    return metas[mes] || { faturamento: 0, lucro: 0, dias: 0 };
  }
  listarTodas() { return this.storage.getObj(this.KEY); }
  /** Meta diária = meta mensal / dias trabalhados planejados */
  metaDiaria(mes) {
    const m = this.porMes(mes);
    return {
      faturamento: m.dias > 0 ? m.faturamento / m.dias : 0,
      lucro: m.dias > 0 ? m.lucro / m.dias : 0,
      dias: m.dias
    };
  }
}

class ToastService {
  constructor(containerId) { this.container = document.getElementById(containerId); }
  show(msg, type = 'success') {
    const el = document.createElement('div');
    el.className = `toast ${type}`;
    el.innerHTML = `<span>${type === 'success' ? '✅' : '❌'}</span> ${msg}`;
    this.container.appendChild(el);
    setTimeout(() => el.remove(), 3200);
  }
}

class AppController {
  constructor() {
    this.storage   = new StorageService();
    this.ganhosSvc = new GanhosService(this.storage);
    this.despSvc   = new DespesasService(this.storage);
    this.metasSvc  = new MetasService(this.storage);
    this.toast     = new ToastService('toastContainer');
    this.tabAtual  = 'ganhos';
  }

  init() {
    document.getElementById('headerDate').textContent =
      new Date().toLocaleDateString('pt-BR', { weekday:'long', day:'2-digit', month:'long' });

    this._setDefaultDates();

    ['g_uber','g_99','g_outros'].forEach(id => {
      document.getElementById(id).addEventListener('input', () => this.calcularLiveTotal());
    });

    document.getElementById('dash_mes').value = Utils.mesAtual();
    document.getElementById('dashMonthLabel').textContent = Utils.mesAnoFormatado(Utils.mesAtual());

    this.renderListaGanhos();
    this.renderListaAbastecimentos();
    this.renderListaDespesas();
    this.renderListaMetas();
    this.renderDashboard();

    // Sync mobile nav active state
    this._syncNav('ganhos');
  }

  _setDefaultDates() {
    const hoje = Utils.hoje();
    ['g_data','ab_data','d_data'].forEach(id => {
      const el = document.getElementById(id);
      if (el) el.value = hoje;
    });
    document.getElementById('m_mes').value = Utils.mesAtual();
  }

  navigate(tab) {
    document.querySelectorAll('.tab-section').forEach(s => s.classList.remove('active'));
    document.getElementById(`tab-${tab}`).classList.add('active');
    this._syncNav(tab);
    this.tabAtual = tab;
    if (tab === 'dashboard') this.renderDashboard();
  }

  _syncNav(tab) {
    document.querySelectorAll('.nav-item').forEach(b => {
      b.classList.toggle('active', b.dataset.tab === tab);
    });
  }

  calcularLiveTotal() {
    const uber   = parseFloat(document.getElementById('g_uber').value)   || 0;
    const n99    = parseFloat(document.getElementById('g_99').value)    || 0;
    const outros = parseFloat(document.getElementById('g_outros').value) || 0;
    const total = uber + n99 + outros;
    document.getElementById('liveTotalValue').textContent = Utils.brl(total);
    const parts = [];
    if (uber   > 0) parts.push(`Uber ${Utils.brl(uber)}`);
    if (n99    > 0) parts.push(`99 ${Utils.brl(n99)}`);
    if (outros > 0) parts.push(`Outros ${Utils.brl(outros)}`);
    document.getElementById('liveTotalSub').textContent =
      parts.length ? parts.join(' + ') : 'Preencha os campos para calcular';
  }

  /** Calcula e exibe as metas diárias automaticamente */
  calcularMetaDiaria() {
    const fat  = parseFloat(document.getElementById('m_faturamento').value) || 0;
    const lucro = parseFloat(document.getElementById('m_lucro').value) || 0;
    const dias = parseInt(document.getElementById('m_dias').value) || 0;

    const result = document.getElementById('metaDiariaResult');
    if (dias > 0 && (fat > 0 || lucro > 0)) {
      result.style.display = 'block';
      document.getElementById('metaDiariaFaturamento').textContent = fat > 0 ? Utils.brl(fat / dias) : 'R$ —';
      document.getElementById('metaDiariaLucro').textContent = lucro > 0 ? Utils.brl(lucro / dias) : 'R$ —';
    } else {
      result.style.display = 'none';
    }
  }

  /* ── GANHOS ── */
  salvarGanho() {
    const data = document.getElementById('g_data').value;
    if (!data) return this.toast.show('Informe a data.', 'error');
    const uber   = parseFloat(document.getElementById('g_uber').value)   || 0;
    const n99    = parseFloat(document.getElementById('g_99').value)    || 0;
    const outros = parseFloat(document.getElementById('g_outros').value) || 0;
    if (uber + n99 + outros === 0) return this.toast.show('Informe ao menos um valor de ganho.', 'error');

    this.ganhosSvc.salvar({
      data,
      ganhoUber: uber, ganho99: n99, ganhoOutros: outros,
      corridasUber:   parseInt(document.getElementById('c_uber').value)  || 0,
      corridas99:     parseInt(document.getElementById('c_99').value)    || 0,
      corridasOutros: parseInt(document.getElementById('c_outros').value)|| 0,
      horas:   parseFloat(document.getElementById('r_horas').value)   || 0,
      kmTrab:  parseFloat(document.getElementById('r_km_trab').value) || 0,
      kmPart:  parseFloat(document.getElementById('r_km_part').value) || 0,
    });

    ['g_uber','g_99','g_outros','c_uber','c_99','c_outros','r_horas','r_km_trab','r_km_part']
      .forEach(id => document.getElementById(id).value = '');
    this.calcularLiveTotal();
    this.renderListaGanhos();
    this.toast.show('Lançamento salvo! 🚀');
  }

  renderListaGanhos() {
    const container = document.getElementById('listaGanhos');
    const list = this.ganhosSvc.listar().slice(0, 15);
    if (!list.length) {
      container.innerHTML = `<div class="empty-state"><div class="emoji">📋</div><p>Nenhum lançamento ainda.</p></div>`;
      return;
    }
    container.innerHTML = list.map(g => {
      const total = (parseFloat(g.ganhoUber)||0) + (parseFloat(g.ganho99)||0) + (parseFloat(g.ganhoOutros)||0);
      const corridas = (parseInt(g.corridasUber)||0) + (parseInt(g.corridas99)||0) + (parseInt(g.corridasOutros)||0);
      return `<div class="list-item">
        <div class="list-item-icon">💰</div>
        <div class="list-item-content">
          <div class="list-item-title">${Utils.dataFormatada(g.data)}</div>
          <div class="list-item-subtitle">${corridas} corridas · ${g.horas || 0}h · ${g.kmTrab || 0} km</div>
        </div>
        <div class="list-item-value" style="color:var(--accent-2)">${Utils.brl(total)}</div>
        <button class="list-item-delete" onclick="App.removerGanho('${g.id}')">✕</button>
      </div>`;
    }).join('');
  }

  removerGanho(id) {
    this.ganhosSvc.remover(id);
    this.renderListaGanhos();
    this.toast.show('Lançamento removido.');
  }

  /* ── ABASTECIMENTO ── */
  salvarAbastecimento() {
    const data = document.getElementById('ab_data').value;
    const veiculo = document.getElementById('ab_veiculo').value;
    const litros = parseFloat(document.getElementById('ab_litros').value);
    const valor = parseFloat(document.getElementById('ab_valor').value);
    const km = parseFloat(document.getElementById('ab_km').value);

      return this.toast.show('Preencha todos os campos.', 'error');

    if (calc) {
      document.getElementById('kmLResult').style.display = 'block';
      document.getElementById('kmLValue').textContent = Utils.dec(calc.media);
      document.getElementById('kmLContext').textContent =
        `${calc.kmRodado.toFixed(0)} km com ${Utils.dec(calc.litrosRef)}L do abastecimento anterior`;
    }

    this.despSvc.salvarAbastecimento({ data, veiculo, litros, valor, km });
    ['ab_litros','ab_valor','ab_km'].forEach(id => document.getElementById(id).value = '');

    this.toast.show('Abastecimento registrado! ⛽');
  }

  removerAbastecimento(id) {

  /* ── METAS ── */

    const mes = document.getElementById('m_mes').value;

    const dias = document.getElementById('m_dias').value;
    if (!mes || (!fat && !lucro)) return this.toast.show('Informe o mês e ao menos uma meta.', 'error');

        <div class="card" style="margin-bottom:10px;">
          <div style="display:flex;align-items:center;justify-content:space-between;margin-bottom:12px;">

  /* ── DASHBOARD ── */
  renderDashboard() {
    const mes = document.getElementById('dash_mes').value || Utils.mesAtual();
    document.getElementById('dashMonthLabel').textContent = Utils.mesAnoFormatado(mes);

    const gTotais  = this.ganhosSvc.totaisMes(mes);
    const dTotais  = this.despSvc.totalDespesasMes(mes);
    const metas    = this.metasSvc.porMes(mes);
    const metaDiaria = this.metasSvc.metaDiaria(mes);

    const faturamentoBruto = gTotais.faturamentoBruto;
    const totalDespesas    = dTotais.total;
    const lucroLiquido     = faturamentoBruto - totalDespesas;
    const margemLucro      = faturamentoBruto > 0 ? (lucroLiquido / faturamentoBruto) * 100 : 0;

    /* Progresso das metas com lógica de ritmo diário */
    const hoje = Utils.hoje();
    const [y, m] = mes.split('-').map(Number);
    const diasNoMes = new Date(y, m, 0).getDate();
    const diaAtual = mes <= hoje.slice(0,7) ? Math.min(parseInt(hoje.slice(8,10)), diasNoMes) : 0;
    const diasRestantes = diasNoMes - diaAtual;

    // Ritmo necessário: quanto precisa faturar por dia nos dias restantes para bater a meta
    const faltaFaturamento = Math.max(0, metas.faturamento - faturamentoBruto);
    const faltaLucro       = Math.max(0, metas.lucro - lucroLiquido);

    // Ritmo ideal baseado na proporção do mês (meta * (diaAtual/diasNoMes))
    const metaEsperadaAgora = metas.faturamento > 0 ? (metas.faturamento * diaAtual / diasNoMes) : 0;
    const ritmoAtual = metaEsperadaAgora > 0 ? (faturamentoBruto / metaEsperadaAgora) * 100 : 0;

    // Progresso real vs meta
    const pctFaturamento = metas.faturamento > 0 ? Utils.clampPct((faturamentoBruto / metas.faturamento) * 100) : 0;
    const pctLucro       = metas.lucro > 0 ? Utils.clampPct((lucroLiquido / metas.lucro) * 100) : 0;

    // Progresso esperado (proporcional ao dia do mês)
    const pctEsperado = Utils.clampPct((diaAtual / diasNoMes) * 100);

    // Ritmo: se estiver acima do esperado = no caminho certo
    let ritmoLabel = '—', ritmoCls = 'no-data';
    if (metas.faturamento > 0 && diaAtual > 0) {
      if (ritmoAtual >= 100) { ritmoLabel = '✅ No ritmo certo'; ritmoCls = 'on-track'; }
      else if (ritmoAtual >= 80) { ritmoLabel = '⚡ Ritmo razoável'; ritmoCls = 'ahead'; }
      else { ritmoLabel = '⚠️ Ritmo abaixo'; ritmoCls = 'behind'; }
    }

    const rPorKm   = gTotais.totalKmTrab > 0 ? faturamentoBruto / gTotais.totalKmTrab : 0;
    const rPorHora = gTotais.totalHoras > 0   ? faturamentoBruto / gTotais.totalHoras  : 0;

    /* ── Tabela de Acompanhamento Diário ── */
    const diasPassados = Utils.diasDoMes(mes);
    let tabelaHtml = '';

    if (metaDiaria.faturamento > 0 && diasPassados.length > 0) {
      const rows = diasPassados.map(dia => {
        const faturadoDia = this.ganhosSvc.faturamentoPorDia(dia);
        const isHoje = dia === Utils.hoje();
        const [, , dd] = dia.split('-');
        const isWeekend = [0, 6].includes(new Date(dia).getDay());

        let statusHtml = '';
        if (faturadoDia === 0) {
          statusHtml = `<span class="status-badge pending">— Sem registro</span>`;
        } else if (faturadoDia >= metaDiaria.faturamento) {
          statusHtml = `<span class="status-badge ok">🟢 Meta batida</span>`;
        } else {
          const faltou = metaDiaria.faturamento - faturadoDia;
          statusHtml = `<span class="status-badge fail">🔴 Faltou ${Utils.brl(faltou)}</span>`;
        }

        return `
          <tr class="${isHoje ? 'today-row' : ''}" style="${isWeekend ? 'opacity:0.6;' : ''}">
            <td class="td-date">${dd}/${String(m).padStart(2,'0')}</td>
            <td class="td-meta">${Utils.brl(metaDiaria.faturamento)}</td>
            <td class="td-faturado ${faturadoDia === 0 ? 'zero' : ''}">${faturadoDia > 0 ? Utils.brl(faturadoDia) : '—'}</td>
            <td>${statusHtml}</td>
          </tr>`;
      }).join('');

      tabelaHtml = `
        <div class="daily-table-wrap">
          <div class="daily-table-header">
            <div class="card-icon blue">📅</div>
            <div class="daily-table-title">Acompanhamento Diário</div>
            <div class="daily-table-subtitle">Meta diária: ${Utils.brl(metaDiaria.faturamento)} · ${metaDiaria.dias} dias no mês</div>
          </div>
          <table>
            <thead>
              <tr>
                <th>Data</th>
                <th>Meta do Dia</th>
                <th>Faturado</th>
                <th>Status</th>
              </tr>
            </thead>
            <tbody>${rows}</tbody>
          </table>
        </div>`;
    } else if (diasPassados.length > 0) {
      tabelaHtml = `
        <div class="card" style="text-align:center;padding:24px;margin-bottom:20px;">
          <div style="font-size:28px;margin-bottom:8px;">📅</div>
          <div style="color:var(--text-muted);font-size:13px;">Configure a meta diária (dias de trabalho no mês) para ver o acompanhamento detalhado aqui</div>
        </div>`;
    }

    const content = document.getElementById('dashboardContent');
    content.innerHTML = `

      <!-- KPIs principais -->
      <div class="metric-row">
        <div class="metric-card">
          <div class="metric-label">💰 Faturamento Bruto</div>
          <div class="metric-value blue">${Utils.brl(faturamentoBruto)}</div>
          <div class="metric-sub">${gTotais.totalCorridas} corridas</div>
        </div>
        <div class="metric-card">
          <div class="metric-label">💚 Lucro Líquido</div>
          <div class="metric-value ${lucroLiquido >= 0 ? 'green' : 'red'}">${Utils.brl(lucroLiquido)}</div>
          <div class="metric-sub">após custos</div>
        </div>
        <div class="metric-card">
          <div class="metric-label">💸 Total Despesas</div>
          <div class="metric-value red">${Utils.brl(totalDespesas)}</div>
          <div class="metric-sub">⛽ ${Utils.brl(dTotais.totalAbastecimentos)}</div>
        </div>
        <div class="metric-card">
          <div class="metric-label">📈 Margem de Lucro</div>
          <div class="metric-value ${margemLucro >= 50 ? 'green' : margemLucro >= 25 ? 'orange' : 'red'}">${margemLucro.toFixed(1)}%</div>
          <div class="metric-sub">${margemLucro >= 60 ? '🟢 Excelente' : margemLucro >= 40 ? '🟡 Boa' : margemLucro >= 20 ? '🟠 Regular' : '🔴 Atenção'}</div>
        </div>
      </div>

      <!-- Metas e Ritmo -->
      ${metas.faturamento > 0 || metas.lucro > 0 ? `
      <div style="display:grid;grid-template-columns:2fr 1fr;gap:16px;margin-bottom:20px;">
        <!-- Progresso das metas -->
        <div class="card" style="margin-bottom:0;">
          <div class="card-header">
            <div class="card-icon blue">🎯</div>
            <div class="card-title">Progresso das Metas</div>
            <div style="margin-left:auto;display:flex;align-items:center;gap:8px;">
              <span style="font-size:11px;color:var(--text-muted);">Esperado hoje: ${pctEsperado.toFixed(0)}%</span>
            </div>
          </div>

          ${metas.faturamento > 0 ? `
          <div class="progress-wrap">
            <div class="progress-header">
              <span class="progress-label">🏆 Faturamento Bruto</span>
              <span class="progress-pct">${pctFaturamento.toFixed(1)}%</span>
            </div>
            <!-- Barra dupla: esperado (fundo) + realizado (frente) -->
            <div class="progress-bar-bg" style="position:relative;">
              <div style="position:absolute;top:0;left:0;height:100%;width:${pctEsperado}%;background:rgba(255,255,255,0.06);border-radius:99px;"></div>
              <div class="progress-bar-fill ${pctFaturamento >= pctEsperado ? '' : 'orange'}" style="width:${pctFaturamento}%"></div>
            </div>
            <div class="progress-missing">
              ${faltaFaturamento > 0
                ? `Faltam ${Utils.brl(faltaFaturamento)} · ${diasRestantes > 0 ? 'Precisa de ' + Utils.brl(faltaFaturamento / diasRestantes) + '/dia' : 'último dia'}`
                : `✅ Meta de ${Utils.brl(metas.faturamento)} atingida!`}
            </div>
          </div>` : ''}

          ${metas.lucro > 0 ? `
          <div class="progress-wrap" style="margin-top:14px;">
            <div class="progress-header">
              <span class="progress-label">💚 Lucro Líquido</span>
              <span class="progress-pct" style="color:var(--accent-2)">${pctLucro.toFixed(1)}%</span>
            </div>
            <div class="progress-bar-bg">
              <div class="progress-bar-fill green" style="width:${pctLucro}%"></div>
            </div>
            <div class="progress-missing">
              ${faltaLucro > 0
                ? `Faltam ${Utils.brl(faltaLucro)} · ${diasRestantes > 0 ? 'Precisa de ' + Utils.brl(faltaLucro / diasRestantes) + '/dia' : 'último dia'}`
                : `✅ Meta de ${Utils.brl(metas.lucro)} atingida!`}
            </div>
          </div>` : ''}
        </div>

        <!-- Ritmo Card -->
        <div class="rhythm-card" style="margin-bottom:0;">
          <div class="rhythm-header">
            <div class="rhythm-title">⚡ Ritmo</div>
            <span class="rhythm-pill ${ritmoCls}">${ritmoLabel}</span>
          </div>
          ${metaDiaria.faturamento > 0 ? `
          <div style="margin-bottom:12px;">
            <div style="font-size:11px;color:var(--text-muted);text-transform:uppercase;letter-spacing:.5px;margin-bottom:4px;">Meta Diária</div>
            <div style="font-family:var(--font-display);font-weight:800;font-size:22px;color:var(--accent);">${Utils.brl(metaDiaria.faturamento)}</div>
          </div>
          ${metaDiaria.lucro > 0 ? `
          <div style="margin-bottom:12px;">
